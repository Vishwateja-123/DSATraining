tree = []

def build(ind, st, ed, arr):
    if st == ed:
        tree[ind] = arr[st]
        return

    mid = (st + ed) // 2
    build(2*ind+1, st, mid, arr)
    build(2*ind+2, mid+1, ed, arr)

    tree[ind] = math.gcd(tree[2*ind+1], tree[2*ind+2])

def update(ind, st, ed, arr, index, val):
    if st == ed:
        tree[ind] = val
        return

    mid = (st + ed) // 2

    if index <= mid:
        update(2*ind+1, st, mid, arr, index, val)
    else:
        update(2*ind+2, mid+1, ed, arr, index, val)

    tree[ind] = math.gcd(tree[2*ind+1], tree[2*ind+2])

def find(ind, st, ed, arr, i, j):
    if i > ed or j < st:
        return 0

    if i <= st and ed <= j:
        return tree[ind]

    mid = (st + ed) // 2
    left = find(2*ind+1, st, mid, arr, i, j)
    right = find(2*ind+2, mid+1, ed, arr, i, j)

    return math.gcd(left, right)

        self.prefix = prefix
        self.suffix = suffix
        self.best = best

def merge(a, b):
    if a is None:
        return b
    if b is None:
        return a

    prefix = a.prefix
    suffix = b.suffix

    if a.prefix == a.length:
        prefix = a.length + b.prefix

    if b.suffix == b.length:
        suffix = b.length + a.suffix

    best = max(a.best, b.best, a.suffix + b.prefix)

    return Node(prefix, suffix, best, a.length + b.length)

        self.leftVal = leftVal
        self.rightVal = rightVal
        self.prefix = prefix

def merge(a, b):
    if a is None:
        return b
    if b is None:
        return a

    prefix = a.prefix
    suffix = b.suffix

    if a.leftVal == b.leftVal:
        prefix = a.prefix + b.prefix

    if a.rightVal == b.rightVal:
        suffix = a.suffix + b.suffix

    best = max(a.best, b.best)

    if a.rightVal == b.leftVal:
        best = max(best, a.suffix + b.prefix)

    return Node(
        a.leftVal,
        b.rightVal,
        prefix,
        suffix,
        best
    )

class Node:
    def __init__(self, open, close, pairs):
        self.open = open
        self.close = close
        self.pairs = pairs

def merge(a, b):
    if a is None:
        return b
    if b is None:
        return a

    x = min(a.open, b.close)

    return Node(
        a.open + b.open - x,
        a.close + b.close - x,
        a.pairs + b.pairs + x
    )

def build(ind, st, ed, s):
    if st == ed:
        if s[st] == '(':
            tree[ind] = Node(1, 0, 0)
        else:
            tree[ind] = Node(0, 1, 0)
        return

    mid = (st + ed) // 2

    build(2*ind+1, st, mid, s)
    build(2*ind+2, mid+1, ed, s)

    tree[ind] = merge(tree[2*ind+1], tree[2*ind+2])

def find(ind, st, ed, l, r):
